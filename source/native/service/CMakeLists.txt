cmake_minimum_required(VERSION 3.15)
project(GuardDeskRemoteService LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
foreach(target GuardDeskService GuardDeskServiceOwnedProbe)
  add_executable(${target} service_host.cpp ../remote/cohort.cpp ../remote/ipc_peer.cpp)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(${target} PRIVATE /W4 /WX /permissive- /sdl /guard:cf)
  target_link_options(${target} PRIVATE /DYNAMICBASE /NXCOMPAT /guard:cf)
  target_link_libraries(${target} PRIVATE kernel32 advapi32 shell32 ole32)
endforeach()
target_compile_definitions(GuardDeskServiceOwnedProbe PRIVATE GUARDDESK_OWNED_SERVICE_TEST)
